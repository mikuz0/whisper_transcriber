Шаг 1: Обновите pip и установочные инструменты
bash

pip install --upgrade pip setuptools wheel

Шаг 2: Установите numpy
bash

pip install numpy==1.24.3

Шаг 3: Установите PyTorch (CPU версия)
bash

pip install torch==2.1.2 torchaudio==2.1.2 --index-url https://download.pytorch.org/whl/cpu

Если эта команда не работает (как в прошлый раз), попробуйте:
bash

pip install torch torchaudio --index-url https://download.pytorch.org/whl/cpu

Шаг 4: Установите openai-whisper
bash

pip install openai-whisper

Шаг 5: Установите остальные зависимости
bash

pip install PyQt5==5.15.9 ffmpeg-python==0.2.0 moviepy==1.0.3 pydub==0.25.1 psutil==5.9.6

Шаг 6: Проверьте установку
bash

python -c "import whisper; print('✓ Whisper установлен')"
python -c "import PyQt5; print('✓ PyQt5 установлен')"
python -c "import torch; print(f'✓ PyTorch {torch.__version__} установлен')"
python -c "import psutil; print('✓ psutil установлен')"
